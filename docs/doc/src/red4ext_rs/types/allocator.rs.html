12/// Engine allocator interface; see the SDK's
13/// [allocators](https://github.com/WopsS/RED4ext.SDK/blob/2216befc23924b46996a9335043aebb31ba6a757/include/RED4ext/Memory/Allocators.hpp).
14#[derive(Debug)]
15#[repr(transparent)]
16pub struct IAllocator(red::Memory::IAllocator);
17
18impl IAllocator {
19    /// Releases storage without dropping its contents.
20    ///
21    /// # Safety
22    /// `memory` must be a live allocation from this allocator, with any required
23    /// payload destruction already complete. It must not be used or freed afterward.
24    #[inline]
25    pub unsafe fn free<T>(&self, memory: *mut T) {
26        let mut alloc = AllocationResult {
27            memory: memory as VoidPtr,
28            size: 0,
29        };
30        unsafe {
31            ((*self.0.vtable_).IAllocator_Free)(
32                &self.0 as *const _ as *mut red::Memory::IAllocator,
33                &mut alloc,
34            )
35        }
36    }
37
38    /// Allocates `size` bytes with the requested alignment; returns null on failure.
39    ///
40    /// # Safety
41    /// The size and alignment must be valid for the engine allocator. Initialize
42    /// the allocation before reading it and release it through the same allocator.
43    #[inline]
44    pub unsafe fn alloc_aligned<T>(&self, size: u32, alignment: u32) -> *mut T {
45        unsafe {
46            let result = ((*self.0.vtable_).IAllocator_GetHandle)(
47                &self.0 as *const _ as *mut red::Memory::IAllocator,
48            );
49            let vault = vault_get(result);
50            vault_alloc_aligned(vault, size, alignment).unwrap_or(ptr::null_mut()) as _
51        }
52    }
53}

55/// Owns storage from an SDK
56/// [memory pool](https://github.com/WopsS/RED4ext.SDK/blob/2216befc23924b46996a9335043aebb31ba6a757/include/RED4ext/Memory/Pool.hpp).
57/// Drop destroys `T` and releases storage. `PoolRef<MaybeUninit<T>>` releases only storage.
58#[derive(Debug)]
59pub struct PoolRef<T: Poolable>(*mut T);
60
61impl<T: Poolable> PoolRef<mem::MaybeUninit<T>> {
62    #[inline]
63    pub(super) unsafe fn assume_init(self) -> PoolRef<T> {
64        let res = PoolRef(self.0 as *mut T);
65        mem::forget(self);
66        res
67    }
68}
69
70impl<T: Poolable> ops::Deref for PoolRef<T> {
71    type Target = T;
72
73    #[inline]
74    fn deref(&self) -> &Self::Target {
75        unsafe { &*self.0 }
76    }
77}
78
79impl<T: Poolable> ops::DerefMut for PoolRef<T> {
80    #[inline]
81    fn deref_mut(&mut self) -> &mut Self::Target {
82        unsafe { &mut *self.0 }
83    }
84}
85
86impl<T: Poolable> Drop for PoolRef<T> {
87    #[inline]
88    fn drop(&mut self) {
89        unsafe {
90            ptr::drop_in_place(self.0);
91            pool_free(self.0);
92        }
93    }
94}

135/// A trait with operations for types that can be stored in a pool.
136#[sealed]
137pub trait PoolableOps: Poolable + Sized {
138    /// Allocates memory for `Self`. The resulting value must be initialized before use.
139    fn alloc() -> Option<PoolRef<mem::MaybeUninit<Self>>>;
140}
141
142#[sealed]
143impl<T: Poolable> PoolableOps for T {
144    fn alloc() -> Option<PoolRef<mem::MaybeUninit<Self>>> {
145        let result = unsafe { vault_alloc(T::Pool::vault(), mem::size_of::<T>() as u32)? };
146        (!result.is_null()).then(|| PoolRef(result.cast::<mem::MaybeUninit<Self>>()))
147    }
148}
149
150unsafe fn pool_free<T: Poolable>(memory: *mut T) {
151    let mut alloc = AllocationResult {
152        memory: memory as VoidPtr,
153        size: 0,
154    };
155    unsafe {
156        let free = crate::fn_from_hash!(
157            Memory_Vault_Free,
158            unsafe extern "C" fn(*mut red::Memory::Vault, *mut AllocationResult)
159        );
160        free(T::Pool::vault(), &mut alloc);
161    }
162}

164/// A trait for different types of pools.
165#[sealed]
166pub trait Pool {
167    const NAME: &'static str;
168
169    #[doc(hidden)]
170    fn vault_cache() -> &'static OnceNonZeroUsize;
171
172    fn vault() -> *mut red::Memory::Vault {
173        Self::vault_cache()
174            .get_or_try_init(|| {
175                NonZeroUsize::new(unsafe { vault_get(fnv1a32(Self::NAME)) as _ }).ok_or(())
176            })
177            .expect("should resolve vault")
178            .get() as _
179    }
180}

224pub(super) unsafe fn vault_alloc(vault: *mut red::Memory::Vault, size: u32) -> Option<VoidPtr> {
225    let mut result = AllocationResult::default();
226    unsafe {
227        let alloc = crate::fn_from_hash!(
228            Memory_Vault_Alloc,
229            unsafe extern "C" fn(*mut red::Memory::Vault, *mut AllocationResult, u32)
230        );
231        alloc(vault, &mut result, size as _);
232    };
233    (!result.memory.is_null()).then_some(result.memory)
234}

236pub(super) unsafe fn vault_alloc_aligned(
237    vault: *mut red::Memory::Vault,
238    size: u32,
239    alignment: u32,
240) -> Option<VoidPtr> {
241    let mut result = AllocationResult::default();
242    unsafe {
243        let alloc_aligned = crate::fn_from_hash!(
244            Memory_Vault_AllocAligned,
245            unsafe extern "C" fn(*mut red::Memory::Vault, *mut AllocationResult, u32, u32)
246        );
247        alloc_aligned(vault, &mut result, size as _, alignment as _);
248    };
249    (!result.memory.is_null()).then_some(result.memory)
250}

252#[cold]
253pub(super) unsafe fn vault_get(handle: u32) -> *mut red::Memory::Vault {
254    unsafe {
255        let vault = red::Memory::Vault::Get();
256        vault_get_from_registry(&raw mut (*vault).poolRegistry, handle)
257    }
258}
259
260unsafe fn vault_get_from_registry(
261    registry: *mut red::Memory::PoolRegistry,
262    handle: u32,
263) -> *mut red::Memory::Vault {
264    unsafe {
265        let nodes = RwSpinLockReadGuard::new(
266            ptr::NonNull::new_unchecked(&raw mut (*registry).nodesLock),
267            ptr::NonNull::new_unchecked(&raw mut (*registry).nodes),
268        );
269        let Some(info) = nodes.iter().find(|node| node.handle == handle) else {
270            return ptr::null_mut();
271        };
272        ((*info.storage).allocatorStorage & !7) as _
273    }
274}

282        let caches = [
283            FunctionPool::vault_cache(),
284            PropertyPool::vault_cache(),
285            ScriptPool::vault_cache(),

287        for (i, cache) in caches.iter().enumerate() {

291        }

309        let expected = vault.as_mut_ptr();
